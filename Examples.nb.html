penguins_clean <- penguins %>%
  mutate(
    bill_ratio = bill_length_mm / bill_depth_mm  # create a new variable
  ) %>%
  drop_na(bill_length_mm, bill_depth_mm, bill_ratio)  # removes rows with NAs

Generating Data of a Given Distribution

  1. Generate random (r-) Normally-distributed data (-norm)

# 1. Generate 1000 random numbers from a normal distribution
set.seed(123)  # for reproducibility
x <- rnorm(1000, mean = 0, sd = 1)
  1. Overlay the theoretical density (d-) curve to the Normal distribution (-norm)
# Histogram of data
hist(x, breaks = 30, probability = TRUE,
     main = "Normal Distribution", col = "lightblue")

# Density function overlay
curve(dnorm(x, mean = 0, sd = 1), add = TRUE, col = "red", lwd = 2)

  1. Determine the probability that X ≤ 1.96 (pnorm = inverse of qnorm)
pnorm(1.96, mean = 0, sd = 1)

  1. Find the 97.5% quantile (qnorm = inverse of pnorm)
qnorm(0.975, mean = 0, sd = 1)
